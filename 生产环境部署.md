#### **B. 批量推理服务**
```python
class DDoSDetectionService:
    def __init__(self, model_path):
        self.agent = DoubleQAgent(observation_space_shape=7, action_space_n=4)
        self.agent.load_model(model_path, load_optimizer=False, load_memory=False)
        self.request_count = 0
    
    def predict_batch(self, features_batch):
        """批量预测"""
        results = []
        for features in features_batch:
            action = self.agent.choose_action(features)
            results.append({
                'action': action,
                'action_name': ['接受', '拒绝', '阻止源', '阻止组'][action],
                'confidence': self._get_confidence(features, action)
            })
        return results
    
    def _get_confidence(self, features, action):
        """获取预测置信度"""
        with torch.no_grad():
            state_tensor = torch.from_numpy(features).float().unsqueeze(0)
            q_values = self.agent.q_func(state_tensor).cpu().numpy()[0]
            return float(np.max(q_values))
```

```python
def predict_batch_optimized(self, features_batch):
    """优化的批量预测"""
    batch_tensor = torch.from_numpy(np.array(features_batch)).float()
    
    with torch.no_grad():
        q_values = self.agent.q_func(batch_tensor)
        actions = torch.argmax(q_values, dim=1)
        confidences = torch.max(q_values, dim=1)[0]
    
    return [
        {
            'action': action.item(),
            'action_name': ['接受', '拒绝', '阻止源', '阻止组'][action.item()],
            'confidence': confidence.item()
        }
        for action, confidence in zip(actions, confidences)
    ]
```

```python
class DDoSDetectionError(Exception):
    """自定义异常类"""
    pass

def safe_predict(features):
    try:
        if not isinstance(features, np.ndarray):
            features = np.array(features, dtype=np.float32)
        
        if features.shape != (7,):
            raise DDoSDetectionError(f"特征维度错误: {features.shape}")
        
        if not np.all((features >= 0) & (features <= 1)):
            raise DDoSDetectionError("特征值必须在0-1之间")
        
        return service.predict_single(features)
    
    except Exception as e:
```
